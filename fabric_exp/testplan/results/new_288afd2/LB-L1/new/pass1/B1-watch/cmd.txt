# The box/watcher tests only fail when a device ASSERT can trap, so the watcher has to be
# enabled before pytest starts. This job runs with enable-watcher off, and with it off ASSERT
# compiles to a no-op (tt_metal/hw/inc/api/debug/assert.h), so these tests would pass whether
# or not the fix under test is present. Same four variables CI's enable-watcher sets, poll at
# 30s. Kept in a separate directory so the run above does not pick them up watcher-less.
export TT_METAL_WATCHER=30 TT_METAL_WATCHER_APPEND=1
export TT_METAL_WATCHER_NOINLINE=1 TT_METAL_WATCHER_DISABLE_ETH=1
pytest tests/ttnn/unit_tests/operations/ccl/blackhole_CI/box/watcher -xv --timeout=600
