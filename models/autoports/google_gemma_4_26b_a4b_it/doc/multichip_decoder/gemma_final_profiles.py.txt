import json,os,subprocess,sys,hashlib
from pathlib import Path
root=Path('/workspace/tt-metal'); p=root/'models/autoports/google_gemma_4_26b_a4b_it/doc/multichip_decoder'
for kind,layer in [('sliding',0),('full',5)]:
 out=p/f'profile_final_{kind}';out.mkdir(exist_ok=True)
 command=[str(root/'python_env/bin/python'),'-m','tracy','-r','-p','-v','--op-support-count','100000','--no-op-info-cache','--disable-device-data-dump-to-files','--disable-device-data-push-to-tracy','-o',str(out),'-n','tp4','-m','models.autoports.google_gemma_4_26b_a4b_it.tests.run_multichip_decoder','--tp','4','--layer',str(layer),'--length','4096','--steps','128','--trace','--profile','--prefill-timing-samples','1','--output',str(out/'run.json')]
 env={k:v for k,v in os.environ.items() if not k.startswith('TT_METAL_WATCHER')};env['HF_HUB_OFFLINE']='1'
 print('START',kind,flush=True)
 with (p/f'profile_final_{kind}.log').open('w') as log:
  r=subprocess.run(command,cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT)
 (out/'capture_command.json').write_text(json.dumps(dict(command=command,exit_code=r.returncode,watcher=False,op_support_count=100000),indent=2)+'\n')
 print('END',kind,r.returncode,flush=True)
 if r.returncode:sys.exit(r.returncode)
