"""Standalone real-weight pinned-order projection test, never a decoder gate.

CPU mode imports no TTNN. Independent blocked pair arithmetic and literal
F.linear are compared at every BF16 output. Device mode requires bit-exact
agreement with BOTH, warmed/captured/replayed entirely under runtime audit.
All shapes are warmed before capture; a different prepared logical count runs
eagerly while each trace is live, then changed and repeated inputs replay.
No readback occurs before the first trace replay. Output buffers are allocated
before capture and remain live until after trace release.

The shape-only candidate policy is K512 forQ1 and K1024 forQ2..32. An explicit
--block-size512/1024 is a discriminating override, not a different acceptance
bar. Denormal/NaN/Inf semantics and arbitrary HF CPU backends are not claimed.
"""

import argparse
import hashlib
import json
import sys
import time
from pathlib import Path

import torch
from safetensors import safe_open


def digest(path):
    value = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            value.update(chunk)
    return value.hexdigest()


def tensor_digest(value):
    return hashlib.sha256(value.contiguous().view(torch.uint8).numpy().tobytes()).hexdigest()


def ordered_cpu(value, weight, block):
    """No pair presumming: BF16 products are exact FP32, each add rounds."""
    count = value.shape[-2]
    products = value.reshape(count, 2560).float().transpose(0, 1).unsqueeze(-1) * weight.float().transpose(
        0, 1
    ).unsqueeze(1)
    total = None
    for begin in range(0, 2560, block):
        partial = torch.zeros((count, 640), dtype=torch.float32)
        for index in range(begin, min(begin + block, 2560), 2):
            partial.add_(products[index + 1])
            partial.add_(products[index])
        total = partial if total is None else partial + total
    return total.bfloat16().reshape(1, 1, count, 640)


def compare(expected, actual):
    x, y = expected.double().reshape(-1, 640), actual.double().reshape(-1, 640)
    xc, yc = x - x.mean(-1, keepdim=True), y - y.mean(-1, keepdim=True)
    denominator = (xc.square().sum(-1) * yc.square().sum(-1)).sqrt()
    pcc = (xc * yc).sum(-1) / denominator
    finite = bool(torch.isfinite(y).all() and torch.isfinite(pcc).all())
    return {
        "bf16_exact": torch.equal(expected, actual),
        "different_elements": int((expected != actual).sum()),
        "minimum_row_pcc": pcc.min().item(),
        "maximum_absolute_error": (x - y).abs().max().item(),
        "finite": finite,
    }


def shape_list(text):
    if text == "all":
        return list(range(1, 33))
    values = [int(x) for x in text.split(",")]
    if not values or len(set(values)) != len(values) or any(not 1 <= x <= 32 for x in values):
        raise ValueError("Counts must be unique values1..32 or all")
    return values


@torch.inference_mode()
def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--snapshot", type=Path, required=True)
    parser.add_argument("--source-cache", type=Path, required=True)
    parser.add_argument("--counts", default="1,2,7,31,32")
    parser.add_argument("--starts", default="105568,105584,58528,58531,16736,16748")
    parser.add_argument("--block-size", type=int, choices=(512, 1024))
    parser.add_argument("--cpu-only", action="store_true")
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    torch.set_num_threads(8)
    counts = shape_list(args.counts)
    starts = [int(x) for x in args.starts.split(",")]
    if len(starts) < 2:
        parser.error("At least two starts are required for changed-input trace replay")
    from transformers.models.qwen4_exp import modeling_qwen4_exp as hf

    source = torch.load(args.source_cache, map_location="cpu", mmap=True, weights_only=True)
    assert source["key"]["snapshot"] == str(args.snapshot.resolve())
    assert source["key"]["config_sha256"] == digest(args.snapshot / "config.json")
    assert source["key"]["reference_sha256"] == digest(Path(hf.__file__))
    assert tuple(source["hidden"].shape) == (1, 262144, 2560)
    assert min(starts) >= 0 and max(starts) + max(counts) <= 262144
    index = json.loads((args.snapshot / "model.safetensors.index.json").read_text())["weight_map"]
    key = "model.language_model.layers.3.self_attn.indexer.index_qk_proj.weight"
    with safe_open(args.snapshot / index[key], framework="pt") as shard:
        weight = shard.get_tensor(key).clone()
    assert tuple(weight.shape) == (640, 2560) and weight.dtype == torch.bfloat16
    assert torch.isfinite(weight).all()
    min_weight = weight.abs()[weight != 0].float().min().item()
    cases = {}
    report = {
        "scope": __doc__,
        "device_executed": False,
        "passed": False,
        "stage_acceptance": "not_evaluated",
        "policy": "Q1:block512;Q2..32:block1024;high-low pair updates then ordered partial adds",
        "explicit_block_override": args.block_size,
        "source_key": source["key"],
        "source_cache": str(args.source_cache),
        "weight_key": key,
        "weight_sha256": tensor_digest(weight),
        "torch_version": torch.__version__,
        "torch_cpu_capability": torch.backends.cpu.get_cpu_capability(),
        "cpu": [],
        "device": [],
    }
    for count in counts:
        block = args.block_size or (512 if count == 1 else 1024)
        cases[count] = []
        for start in starts:
            value = source["hidden"][:, start : start + count].clone().reshape(1, 1, count, 2560)
            assert torch.isfinite(value).all()
            min_value = value.abs()[value != 0].float().min().item()
            assert (
                min_value * min_weight >= torch.finfo(torch.float32).tiny
            ), "Probe does not cover subnormal product semantics"
            literal = torch.nn.functional.linear(value, weight)
            ordered = ordered_cpu(value, weight, block)
            result = compare(literal, ordered)
            cases[count].append((value, literal, ordered))
            report["cpu"].append(
                {
                    "count": count,
                    "start": start,
                    "block_size": block,
                    "input_sha256": tensor_digest(value),
                    "literal_sha256": tensor_digest(literal),
                    "ordered_sha256": tensor_digest(ordered),
                    **result,
                }
            )
        print(
            "ORDERED_CPU",
            count,
            "different",
            sum(row["different_elements"] for row in report["cpu"] if row["count"] == count),
            flush=True,
        )
    args.output.parent.mkdir(parents=True, exist_ok=True)

    def save():
        args.output.write_text(json.dumps(report, indent=2) + "\n")

    report["cpu_order_matches_literal"] = all(row["bf16_exact"] for row in report["cpu"])
    save()
    if args.cpu_only:
        assert "ttnn" not in sys.modules
        assert report[
            "cpu_order_matches_literal"
        ], "Candidate order differs from literal HF; see saved per-shape evidence"
        return

    import ttnn
    from models.autoports.qwen_qwen3_8_flash_next.tests.probe_linear_contract import runtime_audit
    from models.autoports.qwen_qwen3_8_flash_next.tt.ordered_projection import ordered_indexer_projection

    mesh = ttnn.open_mesh_device(ttnn.MeshShape(1, 1), trace_region_size=0)
    trace = None
    report["device_executed"] = True
    try:
        weights = ttnn.from_torch(
            weight.transpose(0, 1).contiguous().reshape(1, 1, 2560, 640),
            dtype=ttnn.bfloat16,
            layout=ttnn.TILE_LAYOUT,
            device=mesh,
            memory_config=ttnn.DRAM_MEMORY_CONFIG,
        )
        inputs, outputs, refresh = {}, {}, {}
        for count in counts:
            refresh[count] = [
                ttnn.from_torch(case[0], dtype=ttnn.bfloat16, layout=ttnn.TILE_LAYOUT) for case in cases[count]
            ]
            inputs[count] = ttnn.to_device(refresh[count][0], mesh)
            outputs[count] = ttnn.empty(
                (1, 1, count, 640),
                dtype=ttnn.bfloat16,
                layout=ttnn.TILE_LAYOUT,
                device=mesh,
                memory_config=ttnn.DRAM_MEMORY_CONFIG,
            )
            with runtime_audit():
                ordered_indexer_projection(inputs[count], weights, output=outputs[count], block_size=args.block_size)
            ttnn.synchronize_device(mesh)
            begin = time.perf_counter()
            with runtime_audit():
                ordered_indexer_projection(inputs[count], weights, output=outputs[count], block_size=args.block_size)
            ttnn.synchronize_device(mesh)
            warmed_host_ms = (time.perf_counter() - begin) * 1000
            actual = ttnn.to_torch(outputs[count])
            metrics = compare(cases[count][0][2], actual)
            report["device"].append(
                {
                    "count": count,
                    "phase": "warm",
                    "warmed_host_wall_ms": warmed_host_ms,
                    "timing_scope": "Diagnostic only; includes Python/runtime-audit/dispatch/synchronization, not a device perf report",
                    **metrics,
                }
            )
            save()
            assert metrics["bf16_exact"], (count, metrics)
        prepared_programs = mesh.num_program_cache_entries()
        for count in counts:
            trace = ttnn.begin_trace_capture(mesh, cq_id=0)
            with runtime_audit():
                ordered_indexer_projection(inputs[count], weights, output=outputs[count], block_size=args.block_size)
            ttnn.end_trace_capture(mesh, trace, cq_id=0)
            # Existing output buffers predate capture; no captured allocation
            # acknowledgement or pre-replay tensor read is necessary.
            other = next((n for n in counts if n != count), count)
            with runtime_audit():
                ordered_indexer_projection(inputs[other], weights, output=outputs[other], block_size=args.block_size)
            first_actual = None
            for case_index in [*range(len(cases[count])), 0]:
                ttnn.copy_host_to_device_tensor(refresh[count][case_index], inputs[count])
                with runtime_audit():
                    ttnn.execute_trace(mesh, trace, cq_id=0, blocking=True)
                actual = ttnn.to_torch(outputs[count])
                literal = compare(cases[count][case_index][1], actual)
                ordered = compare(cases[count][case_index][2], actual)
                report["device"].append(
                    {
                        "count": count,
                        "start": starts[case_index],
                        "phase": "trace_replay",
                        "literal": literal,
                        "ordered": ordered,
                        "runtime_audit": True,
                    }
                )
                save()
                assert ordered["bf16_exact"] and literal["bf16_exact"], (count, case_index, ordered, literal)
                if first_actual is None:
                    first_actual = actual.clone()
                elif case_index == 0:
                    assert torch.equal(first_actual, actual)
            ttnn.release_trace(mesh, trace)
            trace = None
            print("ORDERED_DEVICE_PASS", count, flush=True)
        report.update(
            passed=True,
            runtime_audit=True,
            repeated_input_exact=True,
            capture_output_readbacks_before_replay=0,
            prepared_program_cache_entries=prepared_programs,
            final_program_cache_entries=mesh.num_program_cache_entries(),
        )
        assert mesh.num_program_cache_entries() == prepared_programs, "New persistent program after trace preparation"
    except Exception as error:
        report["passed"] = False
        report["error"] = str(error).split("backtrace:")[0]
        raise
    finally:
        save()
        if trace is not None:
            ttnn.release_trace(mesh, trace)
        ttnn.close_mesh_device(mesh)


if __name__ == "__main__":
    main()
